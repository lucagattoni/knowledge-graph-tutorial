# Extended regular expressions, one per line, used by tools/check_public_safe.sh.
# Blank lines and lines starting with '#' are ignored.

# Paths from a developer machine
/Users/[A-Za-z0-9._-]+
/private/tmp
# A Linux home path stands alone (line start, space, quote, bracket) or follows file://. After a host name it is a web address
# and is not matched, so a link to /home/<author>/ on a university site no longer fails the gate.
(^|[^A-Za-z0-9._~/-])/home/[a-z][a-z0-9_-]+/|file:///home/[a-z]

# Credentials
sk-ant-[A-Za-z0-9_-]{10,}
ghp_[A-Za-z0-9]{20,}
github_pat_[A-Za-z0-9_]{20,}
AKIA[0-9A-Z]{16}
-----BEGIN [A-Z ]*PRIVATE KEY-----
(API_KEY|SECRET|TOKEN|PASSWORD)[A-Z_]*[ ]*=[ ]*['"]?[A-Za-z0-9_-]{12,}

# Personal mailbox addresses
[A-Za-z0-9._%+-]+@(gmail|outlook|hotmail|icloud|proton)\.(com|me)
